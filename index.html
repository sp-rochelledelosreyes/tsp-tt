<!DOCTYPE html>

<html lang="en">

<head>

    <meta charset="UTF-8">

    <meta name="viewport" content="width=device-width, initial-scale=1.0">

    <title>Global Team Time Zone Overlap</title>

    <script crossorigin src="https://unpkg.com/react@18/umd/react.production.min.js"></script>

    <script crossorigin src="https://unpkg.com/react-dom@18/umd/react-dom.production.min.js"></script>

    <script src="https://unpkg.com/@babel/standalone/babel.min.js"></script>

    <script src="https://cdn.tailwindcss.com"></script>

</head>

<body>

    <div id="root"></div>

    

    <script type="text/babel">

        const { useState, useEffect } = React;


        const TimeZoneOverlapVisualizer = () => {

          const [currentTime, setCurrentTime] = useState(new Date());

          const [selectedHour, setSelectedHour] = useState(null);

          const [userTimeZone, setUserTimeZone] = useState('');

          const [userLocalTime, setUserLocalTime] = useState('');


          useEffect(() => {

            const timer = setInterval(() => {

              const now = new Date();

              setCurrentTime(now);

              setUserLocalTime(now.toLocaleTimeString('en-US', { 

                hour: '2-digit', 

                minute: '2-digit',

                hour12: false 

              }));

            }, 1000);

            

            const tz = Intl.DateTimeFormat().resolvedOptions().timeZone;

            setUserTimeZone(tz);

            

            return () => clearInterval(timer);

          }, []);


          const timeZones = [

            { 

              name: 'AET (Australian Eastern Time)',

              abbr: 'AET',

              standard: { label: 'AEST', offset: 10 },

              daylight: { label: 'AEDT', offset: 11, start: 'Oct 6, 2024' },

              color: 'bg-amber-500',

              lightColor: 'bg-amber-100',

              borderColor: 'border-amber-500'

            },

            { 

              name: 'CET (Central European Time)',

              abbr: 'CET',

              standard: { label: 'CET', offset: 1 },

              daylight: { label: 'CEST', offset: 2, start: 'Mar 30, 2025' },

              color: 'bg-blue-500',

              lightColor: 'bg-blue-100',

              borderColor: 'border-blue-500'

            },

            { 

              name: 'ET (Eastern Time)',

              abbr: 'ET',

              standard: { label: 'EST', offset: -5 },

              daylight: { label: 'EDT', offset: -4, start: 'Mar 9, 2025' },

              color: 'bg-green-500',

              lightColor: 'bg-green-100',

              borderColor: 'border-green-500'

            },

            { 

              name: 'MT (Mountain Time)',

              abbr: 'MT',

              standard: { label: 'MST', offset: -7 },

              daylight: { label: 'MDT', offset: -6, start: 'Mar 9, 2025' },

              color: 'bg-purple-500',

              lightColor: 'bg-purple-100',

              borderColor: 'border-purple-500'

            }

          ];


          const isDST = (month) => month >= 3 && month < 11;

          const currentMonth = currentTime.getMonth();


          const getTimeForZone = (tz, hour) => {

            const offset = isDST(currentMonth) ? tz.daylight.offset : tz.standard.offset;

            return (hour - offset + 24) % 24;

          };


          const getCurrentTimeForZone = (tz) => {

            const utcHour = currentTime.getUTCHours();

            const utcMinute = currentTime.getUTCMinutes();

            const offset = isDST(currentMonth) ? tz.daylight.offset : tz.standard.offset;

            const hour = (utcHour + offset + 24) % 24;

            return `${hour.toString().padStart(2, '0')}:${utcMinute.toString().padStart(2, '0')}`;

          };


          const isAcceptableHours = (hour) => hour >= 6 && hour < 19;


          const getOverlapCount = (utcHour) => {

            return timeZones.filter(tz => {

              const localHour = getTimeForZone(tz, utcHour);

              return isAcceptableHours(localHour);

            }).length;

          };


          const hours = Array.from({ length: 24 }, (_, i) => i);


          const userTZ = timeZones.find(tz => {

            const offset = isDST(currentMonth) ? tz.daylight.offset : tz.standard.offset;

            const userOffset = -currentTime.getTimezoneOffset() / 60;

            return offset === userOffset;

          });


          const ClockIcon = () => (

            <svg className="w-8 h-8 text-slate-700" fill="none" stroke="currentColor" viewBox="0 0 24 24">

              <circle cx="12" cy="12" r="10" strokeWidth="2"/>

              <path d="M12 6v6l4 2" strokeWidth="2" strokeLinecap="round"/>

            </svg>

          );


          const UsersIcon = () => (

            <svg className="w-5 h-5" fill="none" stroke="currentColor" viewBox="0 0 24 24">

              <path d="M17 21v-2a4 4 0 0 0-4-4H5a4 4 0 0 0-4 4v2" strokeWidth="2"/>

              <circle cx="9" cy="7" r="4" strokeWidth="2"/>

              <path d="M23 21v-2a4 4 0 0 0-3-3.87M16 3.13a4 4 0 0 1 0 7.75" strokeWidth="2"/>

            </svg>

          );


          const MapPinIcon = () => (

            <svg className="w-4 h-4 text-slate-600" fill="none" stroke="currentColor" viewBox="0 0 24 24">

              <path d="M21 10c0 7-9 13-9 13s-9-6-9-13a9 9 0 0 1 18 0z" strokeWidth="2"/>

              <circle cx="12" cy="10" r="3" strokeWidth="2"/>

            </svg>

          );


          const SmallClockIcon = () => (

            <svg className="w-4 h-4 text-slate-600" fill="none" stroke="currentColor" viewBox="0 0 24 24">

              <circle cx="12" cy="12" r="10" strokeWidth="2"/>

              <path d="M12 6v6l4 2" strokeWidth="2" strokeLinecap="round"/>

            </svg>

          );


          return (

            <div className="w-full max-w-7xl mx-auto p-6 bg-gradient-to-br from-slate-50 to-slate-100 rounded-xl shadow-lg">

              <div className="mb-8">

                <div className="flex items-center gap-3 mb-4">

                  <ClockIcon />

                  <h1 className="text-3xl font-bold text-slate-800">Global Team Time Zone Overlap</h1>

                </div>

                

                <div className="flex flex-wrap gap-4 items-center text-sm">

                  <div className="flex items-center gap-2 bg-white px-4 py-2 rounded-lg shadow-sm">

                    <MapPinIcon />

                    <span className="text-slate-600">Your Location:</span>

                    <span className="font-semibold text-slate-800">{userTimeZone}</span>

                  </div>

                  <div className="flex items-center gap-2 bg-white px-4 py-2 rounded-lg shadow-sm">

                    <SmallClockIcon />

                    <span className="text-slate-600">Your Time:</span>

                    <span className="font-semibold text-slate-800">{userLocalTime}</span>

                  </div>

                  {userTZ && (

                    <div className={`flex items-center gap-2 px-4 py-2 rounded-lg shadow-sm ${userTZ.lightColor} ${userTZ.borderColor} border-2`}>

                      <span className="text-slate-700">You're in: <strong>{userTZ.abbr}</strong></span>

                    </div>

                  )}

                </div>

              </div>


              <div className="grid grid-cols-2 md:grid-cols-4 gap-4 mb-8">

                {timeZones.map((tz, idx) => {

                  const currentTimeStr = getCurrentTimeForZone(tz);

                  const currentHour = parseInt(currentTimeStr.split(':')[0]);

                  const currentLabel = isDST(currentMonth) ? tz.daylight.label : tz.standard.label;

                  const isAcceptable = isAcceptableHours(currentHour);

                  const isUserZone = userTZ && userTZ.abbr === tz.abbr;

                  const isDaylightActive = isDST(currentMonth);

                  

                  return (

                    <div key={idx} className={`p-4 rounded-lg border-2 ${tz.borderColor} ${tz.lightColor} ${isUserZone ? 'ring-2 ring-slate-700' : ''} relative`}>

                      {isUserZone && (

                        <div className="absolute top-2 right-2 text-xs font-bold text-slate-700 bg-white px-2 py-1 rounded">

                          YOU

                        </div>

                      )}

                      <div className="text-base font-bold text-slate-800 mb-1">{tz.name}</div>

                      <div className="text-xs text-slate-500 mb-2">

                        Currently: <strong>{currentLabel}</strong>

                        {!isDaylightActive && (

                          <div className="mt-1">DST starts: {tz.daylight.start}</div>

                        )}

                      </div>

                      <div className="text-2xl font-bold text-slate-900">

                        {currentTimeStr}

                      </div>

                      <div className={`text-xs mt-1 ${isAcceptable ? 'text-green-600 font-semibold' : 'text-slate-400'}`}>

                        {isAcceptable ? '● Acceptable Range' : '○ Outside Range'}

                      </div>

                    </div>

                  );

                })}

              </div>


              <div className="bg-white rounded-lg p-6 shadow-md">

                <h2 className="text-xl font-bold text-slate-800 mb-2 flex items-center gap-2">

                  <UsersIcon />

                  24-Hour Overlap View

                </h2>

                <p className="text-sm text-slate-600 mb-4">

                  Acceptable meeting times: 6 AM - 7 PM (excludes late evenings and early mornings)

                </p>

                

                <div className="space-y-3">

                  {timeZones.map((tz, tzIdx) => {

                    const isUserZone = userTZ && userTZ.abbr === tz.abbr;

                    return (

                      <div key={tzIdx} className="flex items-center gap-3">

                        <div className={`w-32 text-sm font-semibold flex-shrink-0 ${isUserZone ? 'text-slate-900' : 'text-slate-700'}`}>

                          {tz.abbr}

                          {isUserZone && <span className="text-xs block text-slate-500">(You)</span>}

                        </div>

                        <div className="flex-1 flex gap-0.5">

                          {hours.map((utcHour) => {

                            const localHour = getTimeForZone(tz, utcHour);

                            const isAcceptable = isAcceptableHours(localHour);

                            const isCurrent = utcHour === currentTime.getUTCHours();

                            

                            return (

                              <div

                                key={utcHour}

                                className={`flex-1 h-10 relative cursor-pointer transition-all ${

                                  isAcceptable ? tz.color : 'bg-slate-200'

                                } ${isCurrent ? 'ring-2 ring-slate-900' : ''} hover:opacity-80`}

                                onMouseEnter={() => setSelectedHour(utcHour)}

                                onMouseLeave={() => setSelectedHour(null)}

                                title={`UTC ${utcHour}:00 = ${tz.abbr} ${localHour}:00`}

                              >

                                {isCurrent && (

                                  <div className="absolute top-0 left-1/2 transform -translate-x-1/2 -translate-y-1 w-0 h-0 border-l-4 border-r-4 border-b-4 border-transparent border-b-slate-900"></div>

                                )}

                              </div>

                            );

                          })}

                        </div>

                      </div>

                    );

                  })}

                  

                  <div className="flex items-center gap-3 mt-2">

                    <div className="w-32 text-xs text-slate-500 flex-shrink-0">UTC →</div>

                    <div className="flex-1 flex gap-0.5">

                      {hours.map((hour) => (

                        <div key={hour} className="flex-1 text-center text-xs text-slate-500">

                          {hour % 3 === 0 ? hour : ''}

                        </div>

                      ))}

                    </div>

                  </div>


                  <div className="flex items-center gap-3 mt-4 pt-4 border-t border-slate-200">

                    <div className="w-32 text-xs font-bold text-slate-700 flex-shrink-0">Team Overlap</div>

                    <div className="flex-1 flex gap-0.5">

                      {hours.map((utcHour) => {

                        const count = getOverlapCount(utcHour);

                        const opacity = count === 0 ? 0.1 : count / 4;

                        

                        return (

                          <div

                            key={utcHour}

                            className={`flex-1 h-8 flex items-center justify-center text-xs font-bold text-white relative`}

                            style={{ 

                              backgroundColor: `rgba(15, 118, 110, ${opacity})`,

                              border: selectedHour === utcHour ? '2px solid #0f766e' : 'none'

                            }}

                          >

                            {count > 0 && count}

                          </div>

                        );

                      })}

                    </div>

                  </div>

                </div>


                {selectedHour !== null && (

                  <div className="mt-6 p-4 bg-slate-50 rounded-lg border border-slate-200">

                    <div className="font-semibold text-slate-800 mb-2">

                      UTC {selectedHour}:00 - {getOverlapCount(selectedHour)} team member(s) in acceptable range

                    </div>

                    <div className="grid grid-cols-2 md:grid-cols-4 gap-2 text-sm">

                      {timeZones.map((tz, idx) => {

                        const localHour = getTimeForZone(tz, selectedHour);

                        const isAcceptable = isAcceptableHours(localHour);

                        const isUserZone = userTZ && userTZ.abbr === tz.abbr;

                        return (

                          <div key={idx} className={`p-2 rounded ${isAcceptable ? tz.lightColor : 'bg-slate-100'} ${isUserZone ? 'ring-2 ring-slate-700' : ''}`}>

                            <div className="font-semibold">

                              {tz.abbr}

                              {isUserZone && <span className="text-xs ml-1">(You)</span>}

                            </div>

                            <div>{localHour.toString().padStart(2, '0')}:00</div>

                            <div className={`text-xs ${isAcceptable ? 'text-green-600' : 'text-slate-400'}`}>

                              {isAcceptable ? '✓ Good time' : '✗ Too early/late'}

                            </div>

                          </div>

                        );

                      })}

                    </div>

                  </div>

                )}

              </div>


              <div className="mt-6 grid grid-cols-2 md:grid-cols-4 gap-4 text-sm">

                {timeZones.map((tz, idx) => (

                  <div key={idx} className="flex items-center gap-2">

                    <div className={`w-4 h-4 rounded ${tz.color}`}></div>

                    <span className="text-slate-700">{tz.name}</span>

                  </div>

                ))}

              </div>

            </div>

          );

        };


        ReactDOM.render(<TimeZoneOverlapVisualizer />, document.getElementById('root'));

    </script>

</body>

</html>
